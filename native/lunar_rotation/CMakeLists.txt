cmake_minimum_required(VERSION 3.16)
project(jx_lunar_rotation_port LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CSPICE_LIBRARY "" CACHE FILEPATH "Full path to official CSPICE or SpiceyPy libcspice.so")
if(NOT EXISTS "${CSPICE_LIBRARY}")
  message(FATAL_ERROR "Set -DCSPICE_LIBRARY to the existing CSPICE library; no automatic downloads")
endif()
add_executable(jx_lunar_native native_benchmark.cpp)
target_compile_options(jx_lunar_native PRIVATE -Wall -Wextra -Wpedantic -fno-fast-math -ffp-contract=off)
target_link_libraries(jx_lunar_native PRIVATE "${CSPICE_LIBRARY}" ${CMAKE_DL_LIBS} m)
enable_testing()
add_test(NAME lunar_native_controls COMMAND jx_lunar_native --selftest)
